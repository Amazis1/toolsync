<script>
  function toolForm() {
    return {
      toolId: "",
      isStorage: false,
      tooltipMessage: "",
      tooltipVisible: false,
      inputState: "",
      validateToolId() {
        const toolId = this.toolId.trim();
        const category =
          document.querySelector('[name="category"]:checked')?.value || "stamp";
        const location =
          document.querySelector('[name="location"]')?.value || "Werk 1";
        const csrf = document.querySelector(
          '[name="csrfmiddlewaretoken"]'
        ).value;
        if (!toolId) {
          this.inputState = "";
          this.tooltipVisible = false;
          this.tooltipMessage = "";
          return;
        }
        const formData = new FormData();
        formData.append("tool_id", toolId);
        formData.append("category", category);
        formData.append("location", location);
        formData.append("is_storage", this.isStorage ? "true" : "");
        fetch("/validate-tool-id/", {
          method: "POST",
          headers: { "X-CSRFToken": csrf },
          body: formData,
        })
          .then((response) =>
            response.json().then((data) => ({ data, status: response.status }))
          )
          .then(({ data, status }) => {
            if (this.isStorage) {
              this.inputState = "success";
              this.tooltipVisible = false;
              this.tooltipMessage = "";
              return;
            }
            if (status === 200 && data.valid) {
              this.inputState = "success";
              this.tooltipVisible = false;
              this.tooltipMessage = "";
            } else {
              this.inputState = "error";
              this.tooltipMessage = data.tooltip || "ID nicht verfügbar";
              this.tooltipVisible = true;
            }
          })
          .catch(() => {
            this.inputState = "error";
            this.tooltipMessage = "Fehler bei der Validierung";
            this.tooltipVisible = true;
          });
      },
    };
  }
</script>
